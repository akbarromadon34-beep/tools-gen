const demoBase = { pc: "https://example.org/login?nftoken=", phone: "https://example.org/unsupported?nftoken=", tv: "https://example.org/tv8?nftoken=" }; document.getElementById("generate").addEventListener("click", () => { const token = document.getElementById("token").value.trim(); if (!token) { alert("Masukkan token demo terlebih dahulu."); return; } if (token.length > 500 || /[\r\n]/.test(token)) { alert("Token demo tidak valid."); return; } const encodedToken = encodeURIComponent(token); const urls = { pc: demoBase.pc + encodedToken, phone: demoBase.phone + encodedToken, tv: demoBase.tv + encodedToken }; for (const device of ["pc", "phone", "tv"]) { const link = document.getElementById(device); link.textContent = urls[device]; link.href = "#"; } document.getElementById("results").classList.remove("hidden"); });
